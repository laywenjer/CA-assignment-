MAIN PROC

    MOV AX,@DATA
    MOV DS,AX

    JMP SHOW_TITLE

; SHOW TITLE
SHOW_TITLE:
    LEA DX,TITLE_MSG
    MOV AH,09H
    INT 21H
    JMP LOGIN_START

; LOGIN START
LOGIN_START:
    LEA DX,LOGIN_MSG
    MOV AH,09H
    INT 21H

    LEA DX,USER_MSG
    MOV AH,09H
    INT 21H

    LEA DX,USERNAME
    MOV AH,0AH
    INT 21H

    LEA DX,PASS_MSG
    MOV AH,09H
    INT 21H

    MOV AH,0AH
    INT 21H

    CMP USERNAME[1],6
    JNE ADMIN_CHECK_FAIL1
    JMP ADMIN_CHECK_U1

ADMIN_CHECK_FAIL1:
    JMP CHECK_MEMBER1

ADMIN_CHECK_U1:
    CMP USERNAME[2],'w'
    JNE ADMIN_CHECK_FAIL2
    JMP ADMIN_CHECK_U2

ADMIN_CHECK_FAIL2:
    JMP CHECK_MEMBER1

ADMIN_CHECK_U2:

    JNE ADMIN_CHECK_FAIL3
    JMP ADMIN_CHECK_U3

ADMIN_CHECK_FAIL3:

ADMIN_CHECK_U3:

    JNE ADMIN_CHECK_FAIL4
    JMP ADMIN_CHECK_U4

ADMIN_CHECK_FAIL4:

ADMIN_CHECK_U4:

    JNE ADMIN_PASSWORD_FAIL4
    JMP ADMIN_PASSWORD_4

ADMIN_PASSWORD_FAIL4:

MEMBER1_U1:

MEMBER1_FAIL2:

MEMBER1_U2:

MEMBER1_FAIL3:

MEMBER1_FAIL4:

MEMBER1_U4:

MEMBER1_FAIL5:

MEMBER1_U5:

MEMBER1_FAIL6:

MEMBER1_U6:

MEMBER1_PFAIL2:

MEMBER3_P3:

ADMIN_START:

    INT 21H

    INT 21H

    JNE ADMIN_TEST2
    JMP ADMIN_BOOK

    JNE ADMIN_TEST3
    JMP ADMIN_HISTORY

ADMIN_TEST3:

ADMIN_TEST4:

    INT 21H

    JMP ADMIN_START

; ADMIN REPORT - SHOW BOOKING STATS
ADMIN_REPORT:
    LEA DX, REPORT_TITLE
    MOV AH, 09H
    INT 21H

    ; Clear old counts in case admin checks report twice
    MOV VENUE_COUNTS[0], 0
    MOV VENUE_COUNTS[1], 0
    MOV VENUE_COUNTS[2], 0
    MOV VENUE_COUNTS[3], 0
    MOV VENUE_COUNTS[4], 0

    MOV SI, 0   ; BOOKINGS index
    MOV DI, 0   ; VENUE_COUNTS index
    MOV CX, 5   ; 5 venues
VENUE_TALLY:
    PUSH CX
    MOV CX, 6   ; 6 time slots per venue
SLOT_TALLY:
    CMP BOOKINGS[SI], 0
    JE NO_TALLY
    INC VENUE_COUNTS[DI]
NO_TALLY:
    INC SI
    LOOP SLOT_TALLY
    
    INC DI      ; Move to next venue tally
    POP CX
    LOOP VENUE_TALLY

    ; Print results
    LEA DX, REP_V1
    MOV AH, 09H
    INT 21H
    MOV AL, VENUE_COUNTS[0]
    CALL PRINT_NUM

    LEA DX, REP_V2
    MOV AH, 09H
    INT 21H
    MOV AL, VENUE_COUNTS[1]
    CALL PRINT_NUM

    LEA DX, REP_V3
    MOV AH, 09H
    INT 21H
    MOV AL, VENUE_COUNTS[2]
    CALL PRINT_NUM

    LEA DX, REP_V4
    MOV AH, 09H
    INT 21H
    MOV AL, VENUE_COUNTS[3]
    CALL PRINT_NUM

    LEA DX, REP_V5
    MOV AH, 09H
    INT 21H
    MOV AL, VENUE_COUNTS[4]
    CALL PRINT_NUM

    JMP ADMIN_START

    CMP USER_TYPE,1
    JNE VENUE_BACK_MEMBER
    JMP VENUE_BACK_ADMIN

VENUE_BACK_MEMBER:
    JMP MEMBER_START

VENUE_BACK_ADMIN:
    JMP ADMIN_START

; TIME MENU
TIME_START:
    LEA DX,TIME_MENU
    MOV AH,09H
    INT 21H

    MOV AH,01H
    INT 21H

    CMP AL,'1'
    JB TIME_BAD
    JMP TIME_CHECK_HIGH

TIME_CHECK_HIGH:
    CMP AL,'6'
    JA TIME_BAD
    JMP TIME_VALID

TIME_BAD:
    LEA DX,INVALID_MSG
    MOV AH,09H
    INT 21H
    JMP TIME_START

TIME_VALID:
    SUB AL,'0'
    MOV TIME_SLOT,AL
    JMP CHECK_SLOT

; CHECK SLOT
CHECK_SLOT:
    MOV AL,VENUE
    DEC AL
    MOV BL,6
    MUL BL
    MOV BL,TIME_SLOT
    DEC BL
    MOV BH,0
    ADD AX,BX
    MOV SI,AX

    MOV AL,BOOKINGS[SI]
    CMP AL,0
    JE  SLOT_AVAILABLE
    JMP SLOT_BOOKED

SLOT_AVAILABLE:
    JMP CONFIRM_START

SLOT_BOOKED:
    LEA DX,BOOKED_MSG
    MOV AH,09H
    INT 21H
    JMP VENUE_START

; CONFIRMATION
CONFIRM_START:
    LEA DX,CONFIRM_MSG
    MOV AH,09H
    INT 21H

    MOV AH,01H
    INT 21H

    CMP AL,'Y'
    JNE CONFIRM_TEST2
    JMP SAVE_BOOKING

CONFIRM_TEST2:
    CMP AL,'y'
    JNE CONFIRM_TEST3
    JMP SAVE_BOOKING

CONFIRM_TEST3:
    CMP AL,'N'
    JNE CONFIRM_TEST4
    JMP CANCEL_BOOKING

CONFIRM_TEST4:
    CMP AL,'n'
    JNE CONFIRM_BAD
    JMP CANCEL_BOOKING

CONFIRM_BAD:
    LEA DX,INVALID_MSG
    MOV AH,09H
    INT 21H
    JMP CONFIRM_START

; SAVE BOOKING
SAVE_BOOKING:
    MOV AL,VENUE
    DEC AL
    MOV BL,6
    MUL BL
    MOV BL,TIME_SLOT
    DEC BL
    MOV BH,0
    ADD AX,BX
    MOV SI,AX

    MOV AL, CURRENT_USER_ID
    MOV BOOKINGS[SI], AL

    LEA DX,BOOK_OK
    MOV AH,09H
    INT 21H

    JMP BOOKING_DONE

; CANCEL BOOKING
CANCEL_BOOKING:
    LEA DX,CANCEL_MSG
    MOV AH,09H
    INT 21H
    JMP BOOKING_DONE

; BOOKING DONE
BOOKING_DONE:
    CMP USER_TYPE,1
    JNE BOOKING_MEMBER
    JMP BOOKING_ADMIN

BOOKING_MEMBER:
    JMP MEMBER_START

BOOKING_ADMIN:
    JMP ADMIN_START

; DISPLAY_BOOKINGS_ALL - show all bookings (admin)
DISPLAY_BOOKINGS_ALL PROC
    MOV CX, 5          ; outer loop: venues
    MOV BH, 1          ; venue counter
VENUE_LOOP_ALL:
    PUSH CX
    MOV CX, 6          ; inner loop: time slots
    MOV BL, 1          ; time counter
TIME_LOOP_ALL:
    ; calculate index = (venue-1)*6 + (time-1)
    MOV AL, BH
    DEC AL
    MOV DL, 6
    MUL DL
    MOV DL, BL
    DEC DL
    ADD AL, DL
    MOV SI, AX
    MOV AL, BOOKINGS[SI]
    CMP AL, 0
    JE  SKIP_ENTRY_ALL
    ; booked, print entry
    CALL PRINT_BOOKING_ENTRY
SKIP_ENTRY_ALL:
    INC BL
    LOOP TIME_LOOP_ALL
    POP CX
    INC BH
    LOOP VENUE_LOOP_ALL
    RET
DISPLAY_BOOKINGS_ALL ENDP

; DISPLAY_BOOKINGS_USER - show only current user's bookings
DISPLAY_BOOKINGS_USER PROC
    MOV CX, 5
    MOV BH, 1
VENUE_LOOP_USER:
    PUSH CX
    MOV CX, 6
    MOV BL, 1
TIME_LOOP_USER:
    MOV AL, BH
    DEC AL
    MOV DL, 6
    MUL DL
    MOV DL, BL
    DEC DL
    ADD AL, DL
    MOV SI, AX
    MOV AL, BOOKINGS[SI]
    CMP AL, CURRENT_USER_ID
    JNE SKIP_ENTRY_USER
    ; match
    CALL PRINT_BOOKING_ENTRY
SKIP_ENTRY_USER:
    INC BL
    LOOP TIME_LOOP_USER
    POP CX
    INC BH
    LOOP VENUE_LOOP_USER
    RET
DISPLAY_BOOKINGS_USER ENDP

; PRINT_BOOKING_ENTRY - prints "Venue: Swimming Pool, Time: 10 AM - 12 PM"
; BH = venue number (1-5), BL = time slot number (1-6)
PRINT_BOOKING_ENTRY PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    PUSH SI

    ; Print "Venue: "
    LEA DX, ENTRY_MSG       ; "Venue: $"
    MOV AH, 09H
    INT 21H

    ; Get venue name offset
    MOV AL, BH
    DEC AL                  ; AL = 0..4
    MOV AH, 0
    MOV SI, AX
    ADD SI, SI              ; *2 for word array
    MOV DX, VENUE_OFFSETS[SI]
    MOV AH, 09H
    INT 21H

    ; Print ", Time: "
    LEA DX, TIME_MSG        ; ", Time: $"
    MOV AH, 09H
    INT 21H

    ; Get time slot name offset
    MOV AL, BL
    DEC AL                  ; AL = 0..5
    MOV AH, 0
    MOV SI, AX
    ADD SI, SI
    MOV DX, TIME_OFFSETS[SI]
    MOV AH, 09H
    INT 21H

    ; Newline
    MOV DL, 10
    MOV AH, 02H
    INT 21H
    MOV DL, 13
    INT 21H

    POP SI
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_BOOKING_ENTRY ENDP

; PROGRAM EXIT
MAIN_EXIT:
    MOV AH,4CH
    MOV AL,00H
    INT 21H

; HELPER FUNCTION (PRINT_NUM) FOR ADMIN REPORT	
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    ; prepare for division
    MOV AH, 0      ; clear out the top half so AX is just our tally
    MOV CL, 10     ; we will divide by 10
    DIV CL         ; AL gets tens, AH gets ones

    MOV BL, AH     ; save the ones securely in BL for a moment
    
    ; print the tens digit
    MOV DL, AL
    ADD DL, '0'
    MOV AH, 02H
    INT 21H

    ; print the ones digit
    MOV DL, BL
    ADD DL, '0'
    MOV AH, 02H
    INT 21H

    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

MAIN ENDP
END MAIN
